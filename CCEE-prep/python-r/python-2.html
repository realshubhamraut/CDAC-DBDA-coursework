10. Functions

Defining Functions

def function_name(parameters):
    """docstring"""
    # code
    return value

# Example
def greet(name):
    """Greets the user"""
    return f"Hello, {name}!"

result = greet("John")  # "Hello, John!"

Parameters vs Arguments

  • Parameters - Variables in function definition
  • Arguments - Values passed when calling

Types of Arguments

# 1. Positional arguments
def add(a, b):
    return a + b
add(2, 3)  # a=2, b=3

# 2. Keyword arguments
add(b=3, a=2)  # named, order doesn't matter

# 3. Default arguments
def greet(name="Guest"):
    return f"Hello, {name}"
greet()        # "Hello, Guest"
greet("John")  # "Hello, John"

# 4. Variable-length arguments (*args)
def sum_all(*args):
    return sum(args)
sum_all(1, 2, 3, 4)  # 10

# 5. Keyword variable-length (**kwargs)
def info(**kwargs):
    for k, v in kwargs.items():
        print(f"{k}: {v}")
info(name="John", age=25)

Argument Order

def func(pos, *args, default=10, **kwargs):
    pass
# Order: positional → *args → default → **kwargs

Return Statement

def divide(a, b):
    if b == 0:
        return None     # early return
    return a / b

# Multiple returns
def get_stats(nums):
    return min(nums), max(nums), sum(nums)

mn, mx, total = get_stats([1,2,3,4,5])

Lambda Functions (Anonymous)

lambda arguments: expression

# Examples
square = lambda x: x ** 2
square(5)  # 25

add = lambda a, b: a + b
add(2, 3)  # 5

# With built-in functions
nums = [1, 2, 3, 4, 5]
list(map(lambda x: x*2, nums))      # [2, 4, 6, 8, 10]
list(filter(lambda x: x%2==0, nums)) # [2, 4]
sorted(names, key=lambda x: len(x)) # sort by length

Scope (LEGB Rule)

# L - Local (inside function)
# E - Enclosing (outer function)
# G - Global (module level)
# B - Built-in (Python built-ins)

x = "global"

def outer():
    x = "enclosing"
    def inner():
        x = "local"
        print(x)  # local
    inner()

# global keyword
def modify():
    global x
    x = "modified"

# nonlocal keyword (for enclosing scope)
def outer():
    x = 10
    def inner():
        nonlocal x
        x = 20
    inner()
    print(x)  # 20
Key: Default arguments are evaluated ONCE at function definition, not each call!

13. Pickling (Serialization)

What is Pickling?

Converting Python objects to byte stream (serialization) for storage or transmission.

pickle Module

import pickle

# Serialize (dump)
data = {"name": "John", "age": 25, "scores": [90, 85, 88]}

# To file
with open("data.pkl", "wb") as f:
    pickle.dump(data, f)

# To bytes
bytes_data = pickle.dumps(data)

# Deserialize (load)
with open("data.pkl", "rb") as f:
    loaded = pickle.load(f)

# From bytes
loaded = pickle.loads(bytes_data)

What Can Be Pickled?

  • Numbers, strings, booleans
  • Lists, tuples, dictionaries, sets
  • Functions defined at module level
  • Classes and instances

pickle vs json

pickle json
Python-specific Language-independent
Binary format Text format
Any Python object Limited types
Not human-readable Human-readable
Security risk Safer
MCQ Tip: Pickling uses binary mode (wb, rb). Never unpickle untrusted data!

15. OOPs - Classes and Objects

Class Definition

class ClassName:
    # Class attribute (shared by all instances)
    count = 0
    
    # Constructor
    def __init__(self, param1, param2):
        # Instance attributes
        self.attr1 = param1
        self.attr2 = param2
        ClassName.count += 1
    
    # Instance method
    def method(self):
        return self.attr1
    
    # String representation
    def __str__(self):
        return f"Object: {self.attr1}"

Creating Objects

class Person:
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def greet(self):
        return f"Hello, I'm {self.name}"

# Create object
p1 = Person("John", 25)
p2 = Person("Jane", 30)

print(p1.name)      # John
print(p1.greet())   # Hello, I'm John

self Keyword

  • Refers to the current instance
  • Must be first parameter of instance methods
  • Not a keyword, just convention (could be any name)

Class vs Instance Attributes

class Dog:
    species = "Canine"  # Class attribute
    
    def __init__(self, name):
        self.name = name  # Instance attribute

d1 = Dog("Buddy")
d2 = Dog("Max")

d1.species          # "Canine"
Dog.species         # "Canine"
d1.name             # "Buddy"
d2.name             # "Max"

Types of Methods

class MyClass:
    # Instance method
    def instance_method(self):
        return self
    
    # Class method
    @classmethod
    def class_method(cls):
        return cls
    
    # Static method
    @staticmethod
    def static_method():
        return "No self or cls"

16. Inheritance

Basic Inheritance

class Parent:
    def __init__(self, name):
        self.name = name
    
    def greet(self):
        return f"Hello from {self.name}"

class Child(Parent):
    def __init__(self, name, age):
        super().__init__(name)  # Call parent constructor
        self.age = age
    
    def info(self):
        return f"{self.name} is {self.age}"

c = Child("John", 25)
c.greet()  # Inherited from Parent
c.info()   # Defined in Child

Types of Inheritance

Type Description
Single One parent
Multiple Multiple parents
Multilevel Chain of inheritance
Hierarchical One parent, multiple children
Hybrid Mix of above

Multiple Inheritance

class A:
    def method_a(self):
        return "A"

class B:
    def method_b(self):
        return "B"

class C(A, B):  # Multiple inheritance
    pass

c = C()
c.method_a()  # "A"
c.method_b()  # "B"

Method Resolution Order (MRO)

class A: pass
class B(A): pass
class C(A): pass
class D(B, C): pass

D.__mro__  # (D, B, C, A, object)
# Uses C3 linearization algorithm

super() Function

class Parent:
    def __init__(self):
        print("Parent init")

class Child(Parent):
    def __init__(self):
        super().__init__()  # Call parent
        print("Child init")

# Output:
# Parent init
# Child init
Key: Python supports multiple inheritance (unlike Java). Use MRO to understand method lookup order.
17. Overloading and Overriding

Method Overriding

Child class redefines parent's method with same name.

class Animal:
    def sound(self):
        return "Some sound"

class Dog(Animal):
    def sound(self):  # Override
        return "Bark"

class Cat(Animal):
    def sound(self):  # Override
        return "Meow"

d = Dog()
d.sound()  # "Bark" (child's method)

Method Overloading

Python doesn't support traditional overloading. Use default arguments or *args.

# Python way - default arguments
def add(a, b=0, c=0):
    return a + b + c

add(1)        # 1
add(1, 2)     # 3
add(1, 2, 3)  # 6

# Using *args
def add(*args):
    return sum(args)

add(1)
add(1, 2)
add(1, 2, 3, 4, 5)

Operator Overloading

class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __add__(self, other):  # + operator
        return Vector(self.x + other.x, self.y + other.y)
    
    def __str__(self):
        return f"({self.x}, {self.y})"

v1 = Vector(1, 2)
v2 = Vector(3, 4)
v3 = v1 + v2  # Uses __add__
print(v3)     # (4, 6)

Common Operator Methods

Operator Method
+ __add__(self, other)
- __sub__(self, other)
* __mul__(self, other)
/ __truediv__(self, other)
== __eq__(self, other)
< __lt__(self, other)
[] __getitem__(self, key)
MCQ Tip: Python doesn't support method overloading by different signatures. Last definition wins!
18. Data Hiding (Encapsulation)

Access Modifiers (Convention)

Convention Meaning Access
name Public Anywhere
_name Protected (convention) Class & subclasses
__name Private (name mangling) Only within class

Name Mangling

class MyClass:
    def __init__(self):
        self.public = "public"
        self._protected = "protected"
        self.__private = "private"

obj = MyClass()
obj.public          # OK
obj._protected      # OK (but shouldn't)
obj.__private       # AttributeError!
obj._MyClass__private  # Works (name mangling)

Property Decorator

class Person:
    def __init__(self, name):
        self._name = name
    
    @property
    def name(self):
        """Getter"""
        return self._name
    
    @name.setter
    def name(self, value):
        """Setter"""
        if not value:
            raise ValueError("Name cannot be empty")
        self._name = value
    
    @name.deleter
    def name(self):
        """Deleter"""
        del self._name

p = Person("John")
print(p.name)      # Getter
p.name = "Jane"    # Setter
del p.name         # Deleter
Key: __var becomes _ClassName__var (name mangling), not truly private!
19. Generators

What is a Generator?

Function that yields values one at a time, maintaining state between calls. Memory efficient.

Generator Function

# Uses yield instead of return
def count_up_to(n):
    i = 1
    while i <= n:
        yield i
        i += 1

# Usage
gen = count_up_to(5)
next(gen)  # 1
next(gen)  # 2
next(gen)  # 3

# In loop
for num in count_up_to(5):
    print(num)  # 1, 2, 3, 4, 5

Generator Expression

# Like list comprehension but with ()
squares = (x**2 for x in range(5))  # generator
squares = [x**2 for x in range(5)]  # list

list(squares)  # [0, 1, 4, 9, 16]

yield vs return

yield return
Pauses function, saves state Ends function
Can yield multiple times Returns once
Returns generator Returns value
Lazy evaluation Immediate evaluation

Practical Example

# Reading large file line by line
def read_large_file(file_path):
    with open(file_path) as f:
        for line in f:
            yield line.strip()

# Memory efficient - only one line in memory at a time
for line in read_large_file("huge.txt"):
    process(line)
MCQ Tip: Generators are iterators that can only be iterated once!
20. Decorators

What is a Decorator?

Function that modifies behavior of another function without changing its code.

Basic Decorator

def my_decorator(func):
    def wrapper():
        print("Before")
        func()
        print("After")
    return wrapper

@my_decorator
def say_hello():
    print("Hello!")

say_hello()
# Output:
# Before
# Hello!
# After

Decorator with Arguments

def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("Before")
        result = func(*args, **kwargs)
        print("After")
        return result
    return wrapper

@my_decorator
def add(a, b):
    return a + b

add(2, 3)  # Returns 5

Common Built-in Decorators

class MyClass:
    @staticmethod
    def static_method():
        pass
    
    @classmethod
    def class_method(cls):
        pass
    
    @property
    def my_property(self):
        return self._value

Practical Decorators

# Timer decorator
import time

def timer(func):
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"{func.__name__} took {end-start:.2f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(2)

# Debug decorator
def debug(func):
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__} with {args}, {kwargs}")
        return func(*args, **kwargs)
    return wrapper

Chaining Decorators

@decorator1
@decorator2
def func():
    pass

# Same as: func = decorator1(decorator2(func))
Key: Decorators are applied bottom-up when chained!
